import { NextResponse } from "next/server";
import nodemailer from "nodemailer";

// Escapes user-supplied text before it is placed in the HTML email, so submitted markup
// cannot change the owner's notification.
function escapeHtml(value: unknown): string {
  return String(value ?? "")
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;")
    .replace(/'/g, "&#39;");
}

export async function POST(req: Request) {
  try {
    const { name, email, subject, message } = await req.json();

    if (!name || !email || !message) {
      return NextResponse.json(
        { error: "Name, Email, and Message are required fields." },
        { status: 400 }
      );
    }

    // Configure the SMTP transport using Gmail and App Password from env
    const transporter = nodemailer.createTransport({
      service: "gmail",
      auth: {
        user: process.env.EMAIL_USER,
        pass: process.env.EMAIL_PASS,
      },
    });

    const safeName = escapeHtml(name);
    const safeEmail = escapeHtml(email);
    const safeSubject = escapeHtml(subject || "");
    const safeMessage = escapeHtml(message);

    // Email options for sending to the site owner
    const mailOptions = {
      from: process.env.EMAIL_USER, // Sent from the authenticated account
      to: process.env.EMAIL_USER, // Delivered to the site owner
      replyTo: email, // If owner clicks "Reply", it goes to the user who filled the form
      subject: `New Contact Form Submission: ${subject || "No Subject"}`,
      text: `
You have received a new message from the Anantorix Contact Form.

Name: ${name}
Email: ${email}
Subject: ${subject || "N/A"}

Message:
${message}
      `,
      html: `
        <div style="font-family: Inter, Helvetica, Arial, sans-serif; background-color: #ffffff; color: #0A1020; padding: 40px 20px;">
          <div style="max-width: 600px; margin: 0 auto; background-color: #F8F9FC; border: 1px solid #D5DAE3; border-radius: 20px; overflow: hidden;">

            <div style="background-color: #0A1F44; padding: 30px 40px; text-align: center;">
              <h2 style="color: #ffffff; margin: 0; font-size: 24px; letter-spacing: 1px;">ANANTORIX</h2>
              <p style="color: rgba(255,255,255,0.8); margin: 5px 0 0 0; font-size: 14px; text-transform: uppercase; letter-spacing: 2px;">New Contact Request</p>
            </div>

            <div style="padding: 40px;">
              <p style="margin: 0 0 10px 0; color: #5F6675; font-size: 12px; text-transform: uppercase; letter-spacing: 1px;">Sender Name</p>
              <p style="margin: 0 0 20px 0; font-size: 18px; font-weight: bold;">${safeName}</p>

              <p style="margin: 0 0 10px 0; color: #5F6675; font-size: 12px; text-transform: uppercase; letter-spacing: 1px;">Email Address</p>
              <p style="margin: 0 0 20px 0; font-size: 16px;">
                <a href="mailto:${safeEmail}" style="color: #5B3FD1;">${safeEmail}</a>
              </p>

              <p style="margin: 0 0 10px 0; color: #5F6675; font-size: 12px; text-transform: uppercase; letter-spacing: 1px;">Subject</p>
              <p style="margin: 0 0 20px 0; font-size: 16px;">${safeSubject || "No Subject provided"}</p>

              <p style="margin: 0 0 10px 0; color: #5F6675; font-size: 12px; text-transform: uppercase; letter-spacing: 1px;">Message</p>
              <div style="background-color: #ffffff; border: 1px solid #D5DAE3; border-radius: 8px; padding: 20px;">
                <p style="margin: 0; font-size: 15px; line-height: 1.6; white-space: pre-wrap;">${safeMessage}</p>
              </div>
            </div>
          </div>
        </div>
      `,
    };

    // Send the email
    await transporter.sendMail(mailOptions);

    return NextResponse.json(
      { message: "Email sent successfully!" },
      { status: 200 }
    );
  } catch (error) {
    console.error("Error sending email:", error);
    return NextResponse.json(
      { error: "Failed to send email. Please try again later." },
      { status: 500 }
    );
  }
}
