"use client";

import { usePathname } from "next/navigation";
import ContactCTA from "./ContactCTA";

export default function LayoutContact() {
  const pathname = usePathname();
  
  // Hidden on /contact (renders its own form) and on the homepage (has its own Final CTA).
  if (pathname === "/contact" || pathname === "/") {
    return null;
  }

  return <ContactCTA />;
}
